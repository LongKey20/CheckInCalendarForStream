const SPREADSHEET_ID = "18KuJwZIKWDgdJeH3yI9gJOKlU2s0Z1kWRXMfHdAK5jI";
const SHEET_PREFIX = "cal_";
const TZ = "Asia/Tokyo";

function doGet(e) {

  const lock = LockService.getScriptLock();
  lock.waitLock(5000);

  try {

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    const dateStr = e.parameter.date;
    const username = e.parameter.username;
    const displayName = e.parameter.displayName;
    const avatar = e.parameter.avatar;

    if (!username) {
      return json({ success: false, error: "missing params" });
    }

    const todayKey = dateStr || Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy-MM");
    const sheet = getOrCreateSheet(ss, todayKey);
    const data = sheet.getDataRange().getDisplayValues();

    let hasAnyToday = false;
    let alreadyUserToday = false;

    const userDates = new Set();

    /* =========================
       scan sheet
    ========================= */
    for (let i = 1; i < data.length; i++) {

      const rowDate = data[i][0];
      const rowUser = data[i][1];

      // user history
      if (rowUser === username) {
        userDates.add(rowDate);
      }

      // global today check
      if (rowDate === dateStr) {

        hasAnyToday = true;

        if (rowUser === username) {
          alreadyUserToday = true;
        }
      }
    }

    // !月曆 指令用
    if (!dateStr) {
      return json({
        success: true,
        showCalendar: true,
        isFirst: false,
        userDates: Array.from(userDates)
      });
    }

    /* =========================
       already signed
    ========================= */
    if (alreadyUserToday) {

      return json({
        success: true,
        showCalendar: false,
        isFirst: false,
        userDates: []
      });
    }

    const isFirst = !hasAnyToday;

    const timestamp = Utilities.formatDate(
      new Date(),
      TZ,
      "yyyy-MM-dd HH:mm:ss"
    );

    /* =========================
       write row
    ========================= */
    sheet.appendRow([
      dateStr,
      username,
      displayName,
      avatar,
      timestamp,
      isFirst ? "YES" : "NO"
    ]);

    userDates.add(dateStr);

    return json({
      success: true,
      showCalendar: true,
      isFirst,
      userDates: Array.from(userDates)
    });

  } catch (err) {

    return json({
      success: false,
      error: err.toString()
    });

  } finally {
    lock.releaseLock();
  }
}

/* =========================
   sheet helper
========================= */
function getOrCreateSheet(ss, dateStr) {

  const parts = dateStr.split("-");

  const sheetName = `${SHEET_PREFIX}${parts[0]}-${parts[1]}`;

  let sheet = ss.getSheetByName(sheetName);

  if (!sheet) {

    sheet = ss.insertSheet(sheetName);

    sheet.appendRow([
      "date",
      "username",
      "displayName",
      "avatar",
      "timestamp",
      "isFirst"
    ]);
  }

  return sheet;
}

/* =========================
   json response
========================= */
function json(obj) {

  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
